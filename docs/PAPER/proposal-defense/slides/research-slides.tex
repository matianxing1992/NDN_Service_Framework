\begin{frame}{Service Discovery: Several Candidates, One Executor}
\begin{columns}[T,onlytextwidth]
\column{.46\linewidth}
\begin{itemize}
\item The User knows the required service.
\item Reachability, capability and willingness may change.
\item Eligible Providers return ACK Data.
\item The User selects one Provider to execute.
\end{itemize}
\column{.51\linewidth}

\end{columns}
\vspace{3mm}\bluebox{Initial NDNSF: discover alternatives and select one Provider per invocation, not multiple collaborating roles. Arrows show the selection process.}
\end{frame}

\begin{frame}{From Invocation to Collaboration: Two Research Scopes}
\begin{block}{Foundation: evaluate discovery and authorization}
The initial prototype selects one Provider. Discovery confidentiality, service permissions and management costs matter even without collaboration.
\end{block}
\begin{block}{Extension: coordinate complementary roles}
Some tasks need several Providers with different responsibilities. Separate calls do not define a shared role assignment or enforce dependencies between them.
\end{block}
\vspace{3mm}\bluebox{Evaluate the invocation foundation; develop and validate the collaboration extension. Existing invocation results do not establish collaboration correctness.}
\end{frame}

\begin{frame}{What NDN Provides---and What It Does Not}
\small
\begin{tabularx}{\linewidth}{p{.39\linewidth}X}
\toprule NDN building block & Remaining application responsibility\\\midrule
Named Data and signatures [1] & Validate signatures under application trust policies, expressible through a Trust Schema [16]\\\midrule
Interest retrieval and caching [1,26] & Enforce task identity, dependencies, and deadlines; caching is not persistent storage\\\midrule
Dataset synchronization / SVS [2,3] & Discover new names, then retrieve Data; define service-message state transitions\\\midrule
NAC [4] / enhanced NAC-ABE [19] & Use content encryption for read access; separately check service-invocation permissions and state\\\bottomrule
\end{tabularx}
\vspace{2mm}\scriptsize NAC: Name-based Access Control; ABE: attribute-based encryption; SVS: State Vector Sync.
\vspace{2mm}\bluebox{\small NDNSF builds on these mechanisms; it does not invent them.}
\end{frame}

\begin{frame}{Three Research Questions}
\begin{itemize}
\item \textbf{RQ1:} How can NDNSF discover willing Providers, keep descriptors confidential from unauthorized readers, and check service permissions before execution, and at what cost?
\item \textbf{RQ2:} How can single-Provider invocation be extended to roles and named dependencies across Providers, whether Data comes from producers or caches?
\item \textbf{RQ3:} When do these mechanisms improve completion, latency, or execution efficiency, and when do their costs outweigh the benefits?
\end{itemize}
\vspace{3mm}\bluebox{RQ1 evaluates the invocation foundation; RQ2 studies its collaboration extension; RQ3 measures both scopes separately.}
\end{frame}

\begin{frame}{Participants and Trust Bootstrap}
\begin{itemize}
\item User: requests a service and selects a Provider; the extension also assigns collaboration roles.
\item Provider: offers and executes a service. A node may play both roles.
\item Controller: administers identities and service permissions approved by the organization.
\item Participants use provisioned trust anchors and validation rules for certificates and Controller material.
\end{itemize}
\vspace{3mm}\bluebox{Unknown runtime candidates are not unregistered authorization identities. The User discovers current participation; the Controller does not choose this task's Providers.}
\end{frame}

\begin{frame}{Single-Provider Invocation: Four Named Data Types}
\small
\begin{tabularx}{\linewidth}{p{.15\linewidth}X}
\toprule Data & Logical purpose\\\midrule
Request & Describe required service and participation constraints; include protected discovery information and Requester Challenge\\\midrule
ACK & Report positive or negative participation; positive ACK supplies selection information and Provider Challenge\\\midrule
Selection & Assign selected execution; provide input/references, challenge response and protected request-scoped keys\\\midrule
Response & Return status, results or result references; the User still validates them before use\\\bottomrule
\end{tabularx}
\vspace{3mm}\bluebox{Sync helps discover names; Interests retrieve Data. The Provider checks Selection before execution; a positive ACK alone does not start the task.}
\end{frame}

\begin{frame}{Request: Discovery Descriptor; Selection: Task Input}
\begin{columns}[T,onlytextwidth]
\column{.48\linewidth}
\begin{block}{Discovery descriptor}
Needed to decide whether to participate: area, deadline, sensor/model and resource requirements.
\end{block}
\column{.48\linewidth}
\begin{block}{Application input}
Needed to execute the selected task: images, detailed processing parameters, or protected input references.
\end{block}
\end{columns}
\vspace{4mm}\bluebox{The boundary follows necessity. A location needed to assess coverage belongs in the protected descriptor; an image needed only for processing can wait.}
\end{frame}

\begin{frame}{Reason 1: Confidential Discovery Before Selection}
\begin{itemize}
\item The User knows the service, but not yet the currently willing Provider identities.
\item An observation area and deadline can reveal intended activity even without the full application input.
\item The User encrypts the descriptor using ABE and the service attribute, without listing Provider identities. [19]
\end{itemize}
\vspace{3mm}\bluebox{Confidential from parties lacking the required decryption capability. Authorized but unselected Providers can read it; names and traffic patterns remain visible.}
\end{frame}

\begin{frame}{Reason 2: Reuse Service-Permission Management}
\begin{itemize}
\item KP-ABE encodes an access policy in each entity's decryption key. [19]
\item NDNSF combines service entitlements in that policy while retaining identity-signing credentials. [28]
\item Challenge checks reuse these permissions for invocation; User permission keys and challenge processing still add cost.
\end{itemize}
\vspace{3mm}\bluebox{DNMP [12] expresses command permissions through role keys and a Trust Schema. Roles can cover several permissions. Compare credential, policy and key updates; reuse alone does not prove lower total cost.}
\end{frame}

\begin{frame}{How NDNSF Checks Service Permissions}
\begin{enumerate}
\item The User places a Requester Challenge in ABE-protected Request content.
\item The Provider returns it in ACK Data; the User checks the value against the active request.
\item The Provider encrypts its challenge in ACK content. The User returns the value in Selection; the Provider checks it before execution.
\end{enumerate}
\vspace{3mm}\bluebox{Both sides also validate signatures under trust rules and check service permissions and request state. Challenge responses show access to protected values; they do not grant permissions or prevent deliberate sharing.}
\end{frame}

\begin{frame}{DNMP: Trust-Schema Command Authorization}
\small
Distributed Network Measurement Protocol (DNMP) [12] authorizes measurement commands:
\begin{enumerate}
\item The Network Key anchors role keys, device keys, and the schema.
\item The client uses target, probe type, arguments, and user role to find an allowed signing key; among matching keys, it chooses the least privileged.
\item The receiving Network Observer Daemon validates the publication under the schema before dispatching the probe.
\end{enumerate}
\vspace{2mm}\bluebox{DNMP also uses publication lifetimes and retained state against replay, assuming loosely synchronized clocks. These are concrete command-authorization mechanisms, not just identity checks.}
\end{frame}

\begin{frame}{DNMP-inspired Authorization versus NDNSF}
\small
\begin{tabularx}{\linewidth}{p{.27\linewidth}XX}
\toprule Design & Execution authorization & Discovery confidentiality\\\midrule
DNMP-inspired [12] & Check role-constrained signatures and NDNSF invocation state & Pair with service-group encryption or ABE\\\midrule
NDNSF ABE-backed & User/Provider check signatures, challenge responses, service permissions and request state & User encrypts descriptor content with service-level ABE\\\bottomrule
\end{tabularx}
\vspace{3mm}\bluebox{Proposed comparison: retain NDNSF discovery and Selection, with input/result protection and revocation checks in both designs. Trust Schema [16] alone does not encrypt Content; adapting DNMP authorization is not an original DNMP feature. No measured cost advantage is claimed.}
\end{frame}

\begin{frame}{When Another Authorization Design May Be Better}
\begin{itemize}
\item Known Providers and simple rules: signed invocation plus recipient encryption may be sufficient.
\item A small, stable receiver group: ordinary group-key management may be attractive.
\item Rich command-argument restrictions: DNMP-style rules make those constraints explicit. [12]
\item Many changing entitlements: measure both policy/key updates and credential/rule updates before claiming a management advantage.
\end{itemize}
\vspace{3mm}\bluebox{Compare complete combinations of authorization and encryption. ABE-backed authorization is one design choice, not a universally superior solution.}
\end{frame}

\begin{frame}{Authorization Lifecycle and Costs}
\begin{itemize}
\item The Controller publishes signed permission updates; participants apply them only after retrieval and validation.
\item Request-scoped keys protect input/results. Selection delivers keys to selected Providers; the requesting User retains access.
\item Status checks reject withdrawn permissions; key rotation restricts future encrypted exchanges. Current ABE rotation is Controller-wide. [28]
\end{itemize}
\vspace{3mm}\bluebox{Revocation has costs: measure update propagation, rejection delay and rekey traffic. Disconnected nodes cannot learn updates immediately; previously disclosed keys and plaintext cannot be recalled.}
\end{frame}

\begin{frame}{Threat Model: What Is Trusted?}
\begin{itemize}
\item Trusted: Controller authorization and the User's planning decision.
\item Untrusted delivery: an object may arrive through a producer, forwarder, cache, or repository and still requires validation.
\item Adversarial cases to test: wrong identities, permissions, challenges, request state, roles, dependencies, or execution identities.
\item Not established by signatures/ABE: honest resource reports, correct inference, truthful images, or resistance to authorized key sharing.
\end{itemize}
\bluebox{No Byzantine-coordinator consensus claim; no traffic-analysis or instantaneous-revocation guarantee.}
\end{frame}

\begin{frame}{RQ2 Extension: From Positive ACKs to a Role Plan}
\begin{enumerate}
\item Application supplies roles, dependency template, constraints and deadline; the template may be handwritten.
\item The User closes the ACK window and freezes the eligible, validated positive ACKs.
\item The User applies a replaceable planning policy and signs the resulting role assignments and dependencies.
\item Each Provider validates the plan, derives permitted inputs and rechecks its resources before starting.
\end{enumerate}
\vspace{3mm}\bluebox{The snapshot freezes observed candidates for this decision, not every Provider in the network or their future availability.}
\end{frame}

\begin{frame}{A Plan Is More Than a Request ID}
\small
\begin{tabularx}{\linewidth}{>{\raggedright\arraybackslash}p{.3\linewidth}X}
\toprule Existing request ID & Proposed task-specific plan rules\\\midrule
Identifies the request; requires a validated message & Assigns exact Providers to collection and processing roles\\\midrule
Does not alone specify predecessors & Declares which producer role may supply each named dependency\\\midrule
Does not alone describe key scope & Associates permitted objects, execution identity, and authorization/key scope\\\bottomrule
\end{tabularx}
\vspace{4mm}\bluebox{Example: within the same request, a service-authorized UAV that was not assigned ``view 1'' must not satisfy the view-1 dependency.}
\end{frame}

\begin{frame}{Plan-to-Data Enforcement: Check Before Advancing}
\begin{enumerate}
\item Validate the Data signature using the appropriate trust rules.
\item Check active request, plan, execution identity and deadline.
\item Verify that the signer is the assigned predecessor Provider.
\item Check that the object matches the declared edge, permitted name, data format and key scope.
\item Require complete validated input before the dependent computation starts.
\end{enumerate}
\vspace{3mm}\bluebox{Apply the same rules to direct, cached and repository-fetched Data. This is a proposed acceptance contract; full end-to-end qualification remains required.}
\end{frame}

\begin{frame}{Collaboration Recovery Needs Explicit State}
\small
\begin{tabularx}{\linewidth}{p{.3\linewidth}X}
\toprule Event & Required behavior\\\midrule
Late ACK & Do not silently change the committed plan\\\midrule
Duplicate Selection & Follow recorded one-time state; do not implicitly repeat a side effect\\\midrule
Incomplete dependency & Wait within policy/deadline, recover, or fail; do not execute on partial data\\\midrule
Provider replacement & Authorize a new assignment; reject old execution actions and specify permitted result reuse\\\midrule
Cancellation & Prevent late messages from reauthorizing cancelled work\\\bottomrule
\end{tabularx}
\vspace{3mm}\bluebox{Network re-retrieval is not task re-execution. Compensation is not an atomic transaction or exactly-once physical action.}
\end{frame}

\begin{frame}{Proposed UAV Collaboration: Capture, Analyze and Return}
\small An operator requests ground-area views from unmanned aerial vehicles (UAVs), followed by analysis.
\begin{enumerate}
\item Sensing UAVs sign encrypted ViewImage Data under their authorized namespaces.
\item The compute Provider retrieves images over assigned edges and publishes Annotation Data.
\item Its encrypted result gives the requester original image names and permitted image keys.
\item The requester retrieves the original images and displays the annotations.
\end{enumerate}
\vspace{2mm}\bluebox{Original image Data remains under its producer's name. GPS alone does not establish camera visibility.}
\end{frame}

\begin{frame}{What the UAV Case Must Demonstrate}
\begin{itemize}
\item A complete authorized collection--processing--return task before the deadline.
\item Correct handling of unavailable views, delayed images and a failed processing Provider.
\item Reject wrong-role/edge inputs; accept cached images only when they satisfy the same task checks.
\item MiniNDN [22] and PX4 SITL [23] results clearly distinguished from physical-flight evidence.
\end{itemize}
\vspace{3mm}\bluebox{Higher recognition/localization accuracy is a separate claim requiring ground truth and a single-view control. It is not assumed here.}
\end{frame}

\begin{frame}{Distributed Inference: Named Computation Dependencies}
\begin{center}
\begin{tikzpicture}[box/.style={draw=MemphisBlue,fill=LightBand,rounded corners,text width=30mm,minimum height=12mm,align=center,font=\small},>=Stealth]
\node[box] (a) {Provider A\\layer group 1};

\draw[->,thick] (a)--node[above,font=\scriptsize]{activation} (b);
\draw[->,thick] (b)--node[above,font=\scriptsize]{activation} (c);
\end{tikzpicture}
\end{center}
\begin{itemize}
\item Pipeline: sequential layers/groups; tensor partitioning additionally needs supported collectives.
\item ONNX [25] describes the graph; export alone does not implement distributed execution.
\item Repo [24] stores artifacts; forwarding caches do not guarantee persistence.
\end{itemize}
\bluebox{The framework contribution is checked runtime assignment and data exchange, not model partitioning itself.}
\end{frame}

\begin{frame}{DI State and Evidence Boundaries}
\begin{itemize}
\item Model weights, reusable artifacts, request activations and KV state have different lifetimes.
\item With KV caching, the runtime reuses prior key/value (K/V) tensors and supplies next-step inputs; resident weights do not replace this state. [31]
\item An in-process test does not establish correct execution across processes or devices.
\item Freeze one supported model/partition first; tensor/hybrid and LLM streaming need their own evidence if claimed.
\end{itemize}
\vspace{3mm}\bluebox{Earlier small-model measurements do not validate the newer distributed inference and recovery paths. [28]}
\end{frame}

\begin{frame}{RQ3: Define the Baselines Before Comparing}
\begin{itemize}
\item NSC [10]: named service calls; describe the tested executor, timeout, queue and recovery behavior.
\item gRPC [17]: describe resolver, sequential/parallel policy, deadlines, retry and cancellation.
\item Match Provider count, capacity, task, topology/traces, publication times and measurement windows.
\item Report fixed-list and dynamically discovered configurations as different configurations.
\end{itemize}
\bluebox{An incomplete baseline cannot establish a limitation of the full framework. Historical unmatched data remain diagnostics.}
\end{frame}

\begin{frame}{Expected Research Outcomes}
\begin{itemize}
\item \textbf{RQ1 answer:} when ABE-backed authorization is preferable to DNMP-inspired role rules plus encryption, accounting for confidentiality, management and withdrawal costs.
\item \textbf{RQ2 answer:} exactly which task-assignment mistakes are rejected and which recovery behavior is supported.
\item \textbf{RQ3 answer:} measured benefits, negative results and deployment limits.
\item Reproducible configurations and evidence linking each conclusion to its tested scope.
\end{itemize}
\bluebox{The outcome is a tested service protocol and an evidence-backed account of when its mechanisms are worth their cost.}
\end{frame}
